{{- /* Gemini path of a page on the capsule's host, e.g. "/padshell/devlog/first-release/". public/gemini/ is served at params.capsulePath. */ -}} {{- $path := "" -}} {{- with .OutputFormats.Get "gemini" -}} {{- $prefix := path.Join (urls.Parse site.BaseURL).Path "gemini" -}} {{- $rel := strings.TrimPrefix $prefix .RelPermalink | strings.TrimSuffix "index.gmi" -}} {{- $path = printf "%s/%s" (strings.TrimSuffix "/" (site.Params.capsulePath | default "/")) (strings.TrimPrefix "/" $rel) -}} {{- end -}} {{- return $path -}}
